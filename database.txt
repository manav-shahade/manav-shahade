CREATE TABLE Customer (cid INT PRIMARY KEY AUTO_INCREMENT, email VARCHAR(255),password VARCHAR(255),address VARCHAR(255),subsidy DECIMAL(10, 2),name VARCHAR(255), phno VARCHAR(15));
    
   
    
    
    
  
   
    

CREATE TABLE Product (pid INT PRIMARY KEY,pname VARCHAR(255) ,ncost DECIMAL(10, 2),scost DECIMAL(10, 2),pimg VARCHAR(255));
